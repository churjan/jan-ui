import { Step, Steps } from "fumadocs-ui/components/steps"
import { Callout } from "fumadocs-ui/components/callout"
import { ShadcnCommand } from "@/components/command-tabs"
import { githubRepository } from "@/lib/components"

<Callout type="warn" className="ps-3 [&>div[role=none]]:hidden">
  Jan UI 使用 [shadcn/ui](https://ui.shadcn.com/docs/installation) 的安装流程。
</Callout>

<Steps>
<Step>

### 初始化项目

在你的 React 与 Tailwind CSS v4 项目中执行，按提示完成初始化：

<ShadcnCommand command="init" />

所需依赖、主题样式和 `cn` 工具函数会自动配置。

</Step>
<Step>

### 添加组件

以 Copy Button 为例，执行以下命令添加组件及其依赖：

<ShadcnCommand command={`add ${githubRepository}/copy-button`} />

其他组件的安装命令见对应文档。

</Step>
<Step>

### 导入并使用

```tsx
import { CopyButton } from "@/components/jan-ui/copy-button"

export function Example() {
  return <CopyButton text="需要复制的内容" />
}
```

组件源码保存在你的项目中，可以直接修改样式和行为。

</Step>
</Steps>
